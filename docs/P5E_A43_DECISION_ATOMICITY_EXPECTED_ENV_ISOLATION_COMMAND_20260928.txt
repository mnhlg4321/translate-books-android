function Assert-PinnedFile {
    param(
        [Parameter(Mandatory = $true)][string]$Path,
        [Parameter(Mandatory = $true)][AllowEmptyString()][string]$ExpectedSha256,
        [Parameter(Mandatory = $true)][string]$Label
    )
    if ([string]::IsNullOrWhiteSpace($ExpectedSha256) -or $ExpectedSha256 -notmatch '^[0-9a-fA-F]{64}$') { throw ($Label + '_EXPECTED_HASH_INVALID_STOP') }
    $literal = [IO.Path]::GetFullPath($Path)
    if (-not (Test-Path -LiteralPath $literal -PathType Leaf)) { throw ($Label + '_MISSING_STOP') }
    $item = Get-Item -LiteralPath $literal -Force
    if ($item.PSIsContainer) { throw ($Label + '_NOT_REGULAR_STOP') }
    if (($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0) { throw ($Label + '_LEAF_REPARSE_STOP') }
    $resolved = [IO.Path]::GetFullPath((Resolve-Path -LiteralPath $literal).Path)
    if (-not [StringComparer]::OrdinalIgnoreCase.Equals($literal, $resolved)) { throw ($Label + '_PATH_SWAP_STOP') }
    $cursor = [IO.Directory]::GetParent($literal)
    while ($null -ne $cursor) {
        if (-not (Test-Path -LiteralPath $cursor.FullName -PathType Container)) { throw ($Label + '_ANCESTOR_MISSING_STOP') }
        $cursorItem = Get-Item -LiteralPath $cursor.FullName -Force
        if (($cursorItem.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0) { throw ($Label + '_ANCESTOR_REPARSE_STOP') }
        $parent = $cursor.Parent
        if ($null -eq $parent -or $parent.FullName -ceq $cursor.FullName) { break }
        $cursor = $parent
    }
    $actual = (Get-FileHash -LiteralPath $literal -Algorithm SHA256).Hash.ToUpperInvariant()
    if ($actual -cne $ExpectedSha256.ToUpperInvariant()) { throw ($Label + '_HASH_MISMATCH_STOP') }
    return [pscustomobject]@{ Path = $literal; Hash = $actual; Length = [long]$item.Length }
}

}

function ConvertTo-WindowsProcessArgument {
    param([Parameter(Mandatory = $true)][AllowEmptyString()][string]$Value)
    if ($Value.Length -gt 0 -and $Value -notmatch '[\s"]') { return $Value }
    $builder = [Text.StringBuilder]::new()
    [void]$builder.Append('"')
    $slashes = 0
    foreach ($character in $Value.ToCharArray()) {
        if ($character -eq '\') { $slashes++; continue }
        if ($character -eq '"') {
            [void]$builder.Append(('\' * (2 * $slashes + 1))); [void]$builder.Append('"'); $slashes = 0; continue
        }
        if ($slashes -gt 0) { [void]$builder.Append(('\' * $slashes)); $slashes = 0 }
        [void]$builder.Append($character)
    }
    if ($slashes -gt 0) { [void]$builder.Append(('\' * (2 * $slashes))) }
    [void]$builder.Append('"')
    return $builder.ToString()
}

function Set-P5EPhaseEnvironment {
    param(
        [Parameter(Mandatory = $true)][Diagnostics.ProcessStartInfo]$StartInfo,
        [Parameter(Mandatory = $true)][ValidateSet('PrepareEvent','Before','Dispatch','After','Verify')][string]$Phase
    )
    # Clear every authority/expected-value variable first.  The child gets an
    # explicit phase environment; only Dispatch receives the expected value.
    $names = @(
        'P5E_OWNER_DECISION_RECEIPT_PATH',
        'P5E_OWNER_DECISION_RECEIPT_SHA256',
        'P5E_A43_COMMAND_SHA256',
        'P5E_A43_RESERVATION_RECEIPT_SHA256',
        'P5E_OWNER_ENDPOINT_ACCOUNT_FINGERPRINT')
    foreach ($name in $names) {
        $property = $StartInfo.PSObject.Properties['Environment']
        if ($null -ne $property) { [void]$StartInfo.Environment.Remove($name) }
        else { [void]$StartInfo.EnvironmentVariables.Remove($name) }
    }
    if ($Phase -eq 'PrepareEvent') {
        $receiptBinding = [string]$receipt.ReceiptSha256
        if ($receiptBinding -notmatch '^[0-9a-fA-F]{64}$') { throw 'P5E_OWNER_RECEIPT_HASH_BINDING_INVALID_STOP' }
        $property = $StartInfo.PSObject.Properties['Environment']
        if ($null -ne $property) { $StartInfo.Environment['P5E_A43_RESERVATION_RECEIPT_SHA256'] = $receiptBinding.ToLowerInvariant() }
        else { $StartInfo.EnvironmentVariables['P5E_A43_RESERVATION_RECEIPT_SHA256'] = $receiptBinding.ToLowerInvariant() }
    }
    if ($Phase -eq 'Dispatch') {
        $expected = [Environment]::GetEnvironmentVariable('P5E_OWNER_ENDPOINT_ACCOUNT_FINGERPRINT', 'Process')
        if ([string]::IsNullOrWhiteSpace($expected) -or $expected -cnotmatch '^[0-9a-f]{64}$') { throw 'P5E_DISPATCH_EXPECTED_VALUE_MISSING_OR_INVALID_STOP' }
        $property = $StartInfo.PSObject.Properties['Environment']
        if ($null -ne $property) { $StartInfo.Environment['P5E_OWNER_ENDPOINT_ACCOUNT_FINGERPRINT'] = $expected }
        else { $StartInfo.EnvironmentVariables['P5E_OWNER_ENDPOINT_ACCOUNT_FINGERPRINT'] = $expected }
        return 'EXPECTED_PRESENT'
    }
    return 'EXPECTED_ABSENT'
}

function Get-HelperOutcomeToken {
    param([Parameter(Mandatory = $true)]$Result)
    $code = if ($null -eq $Result.ProcessExitCode) { 'UNKNOWN' } else { [string]$Result.ProcessExitCode }
    return ('WRAPPER_' + [string]$Result.WrapperExitCode + '_PROCESS_' + $code + '_' + [string]$Result.WrapperStatus)
}

function Invoke-CheckedHelper {
    param(
        [Parameter(Mandatory = $true)][string[]]$Arguments,
        [Parameter(Mandatory = $true)][ValidateSet('PrepareEvent','Before','Dispatch','After','Verify')][string]$Phase,
        [int]$TimeoutMilliseconds = 300000
    )
    $startInfo = [Diagnostics.ProcessStartInfo]::new()
    $startInfo.FileName = $powershellPath
    $startInfo.UseShellExecute = $false
    $startInfo.CreateNoWindow = $true
    $startInfo.RedirectStandardOutput = $true
    $startInfo.RedirectStandardError = $true
    $phaseEnvironmentStatus = Set-P5EPhaseEnvironment -StartInfo $startInfo -Phase $Phase
    $allArguments = @('-NoLogo', '-NoProfile', '-NonInteractive', '-ExecutionPolicy', 'Bypass', '-File', $helperPath) + $Arguments
    $startInfo.Arguments = [string]::Join(' ', @($allArguments | ForEach-Object { ConvertTo-WindowsProcessArgument -Value ([string]$_) }))
    $process = $null
    $launch = $null
    $launchCount = 0L
    $capture = $null
    $processExitCode = $null
    try {
        $launch = [P5EProcessLauncher]::Start($startInfo)
        $process = $launch.Process
        $launchCount = 1L
        $capture = [P5EProcessTextCapture]::Capture($launch, $stdoutCaptureLimit, $stderrCaptureLimit, $TimeoutMilliseconds, $null)
        if ($null -ne $capture.ExitCode) { $processExitCode = [int]$capture.ExitCode }
    } catch {
        if ($launchCount -eq 0L) { throw 'P5E_HELPER_PROCESS_START_FAILED_STOP' }
    } finally {
        if ($null -ne $launch) { $launch.Dispose() } elseif ($null -ne $process) { $process.Dispose() }
    }
    $wrapperStatus = if ($launchCount -eq 0L) { 'PROCESS_START_FAILED' }
        elseif ($null -eq $capture) { 'PROCESS_CAPTURE_EXCEPTION' }
        elseif (-not [bool]$capture.ContainmentVerified) { 'EXTERNAL_PROCESS_STATE_UNKNOWN' }
        elseif ([bool]$capture.OutputTooLarge) { 'OUTPUT_TOO_LARGE' }
        elseif (-not [string]::IsNullOrWhiteSpace([string]$capture.CaptureErrorClass)) { 'STREAM_CAPTURE_FAILED' }
        elseif (-not [bool]$capture.CaptureBounded) { 'STREAM_CAPTURE_NOT_BOUNDED' }
        elseif ([bool]$capture.TimedOut) { 'TIMEOUT' }
        elseif ($null -eq $processExitCode) { 'PROCESS_EXIT_UNKNOWN' }
        else { 'PROCESS_COMPLETED' }
    $wrapperExitCode = switch ($wrapperStatus) {
        'PROCESS_START_FAILED' { 126; break }
        'PROCESS_CAPTURE_EXCEPTION' { 125; break }
        'EXTERNAL_PROCESS_STATE_UNKNOWN' { 126; break }
        'OUTPUT_TOO_LARGE' { 125; break }
        'STREAM_CAPTURE_FAILED' { 125; break }
        'STREAM_CAPTURE_NOT_BOUNDED' { 125; break }
        'TIMEOUT' { 124; break }
        default { if ($null -eq $processExitCode) { 126 } else { [int]$processExitCode }; break }
    }
    return [pscustomobject]@{
        WrapperExitCode = [int]$wrapperExitCode
        WrapperStatus = $wrapperStatus
        ProcessExitCode = $processExitCode
        StdoutDrainStatus = if ($null -eq $capture) { 'NOT_STARTED' } else { [string]$capture.StdoutDrainStatus }
        StderrDrainStatus = if ($null -eq $capture) { 'NOT_STARTED' } else { [string]$capture.StderrDrainStatus }
        StdoutByteCount = if ($null -eq $capture) { -1L } else { [long]$capture.StdoutByteLength }
        StderrByteCount = if ($null -eq $capture) { -1L } else { [long]$capture.StderrByteLength }
        TimedOut = [bool]($null -ne $capture -and $capture.TimedOut)
        CaptureBounded = [bool]($null -ne $capture -and $capture.CaptureBounded)
        CaptureOverflow = [bool]($null -ne $capture -and $capture.OutputTooLarge)
        ContainmentStatus = if ($null -eq $capture) { 'NOT_STARTED' } else { [string]$capture.ContainmentStatus }
        ContainmentVerified = [bool]($null -ne $capture -and $capture.ContainmentVerified)
        LaunchCount = [long]$launchCount
        Phase = $Phase
        ExpectedEnvironment = $phaseEnvironmentStatus
    }
}

if ($null -eq $afterResult -or $afterResult.WrapperExitCode -ne 0) { throw ('P5E_A43_AFTER_NOT_PROVEN_NO_REDISPATCH_' + $(if ($null -eq $afterResult) { 'UNKNOWN' } else { Get-HelperOutcomeToken -Result $afterResult })) }
if ($verifyResult.WrapperExitCode -ne 0) { throw ('P5E_A43_VERIFY_NOT_PROVEN_NO_REDISPATCH_' + (Get-HelperOutcomeToken -Result $verifyResult)) }

Write-Output ('P5E_A43_EVENT_DIRECTORY=' + $eventDirectory)
Write-Output 'P5E_A43_RESERVATIONS=1'
Write-Output 'P5E_A43_DISPATCHES=1'
Write-Output 'P5E_A43_RETRIES=0'
Write-Output 'P5E_A43_REDISPATCHES=0'
Write-Output 'P5E_A43_P6_READY=FALSE'
} finally {
    Clear-P5EProcessDecisionEnvironment
}
