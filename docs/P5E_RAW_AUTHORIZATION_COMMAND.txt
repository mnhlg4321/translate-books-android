# and verifies it.  Every helper launch is preceded by the command-side hash
# gate and the helper repeats the gate before any environment/device access.
# The collector is read-only and never provider-calls, repairs, retries,

$manifestPath = 'D:\App Translate Books\App Translate Books-translation-profile\docs\P5E_RAW_AUTHORIZATION_APPROVAL_MANIFEST.md'

$testApkPath = 'D:\App Translate Books\App Translate Books-translation-profile\artifacts\test-builds\v4.17-p5e.11\a4-1-test-20260914-065532\app-debug-androidTest.apk'

function Assert-HelperHash {
    param([Parameter(Mandatory = $true)][string]$Path,
          [Parameter(Mandatory = $true)][string]$ExpectedSha256)
    if ($ExpectedSha256 -notmatch '^[0-9a-fA-F]{64}$') { throw 'COMMAND_HELPER_HASH_EXPECTATION_INVALID_STOP' }
    $literalPath = [IO.Path]::GetFullPath($Path)
    if (-not (Test-Path -LiteralPath $literalPath -PathType Leaf)) { throw 'COMMAND_HELPER_MISSING_STOP' }
    $item = Get-Item -LiteralPath $literalPath -Force
    if ($item.PSIsContainer -or (($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0)) {
        throw 'COMMAND_HELPER_NOT_REGULAR_STOP'
    }
    $resolvedPath = [IO.Path]::GetFullPath((Resolve-Path -LiteralPath $literalPath).Path)
    if (-not [StringComparer]::OrdinalIgnoreCase.Equals($literalPath, $resolvedPath)) { throw 'COMMAND_HELPER_PATH_SWAP_STOP' }
    $actual = (Get-FileHash -LiteralPath $literalPath -Algorithm SHA256).Hash.ToLowerInvariant()
    if ($actual -cne $ExpectedSha256.ToLowerInvariant()) { throw 'COMMAND_HELPER_HASH_MISMATCH_STOP' }
}

$powershellPath = (Get-Command powershell.exe -ErrorAction Stop).Source
$eventId = 'raw-live-' + [DateTimeOffset]::UtcNow.ToString('yyyyMMdd-HHmmssfff') + '-' + [Guid]::NewGuid().ToString('N')
$eventDirectory = Join-Path $evidenceRoot $eventId
if (-not (Test-Path -LiteralPath $evidenceRoot -PathType Container)) {
    New-Item -ItemType Directory -Path $evidenceRoot -ErrorAction Stop | Out-Null
}
New-Item -ItemType Directory -Path $eventDirectory -ErrorAction Stop | Out-Null

function Invoke-CheckedHelper {
    param([Parameter(Mandatory = $true)][string[]]$Arguments)
    Assert-HelperHash -Path $helperPath -ExpectedSha256 $expectedHelperSha256
    # Keep helper stdout visible without mixing it into the numeric exit value
    # returned by this wrapper.  The helper writes durable metadata/captures to
    # the event directory; this wrapper never prints its argv.
    & $powershellPath -NoLogo -NoProfile -NonInteractive -ExecutionPolicy Bypass -File $helperPath @Arguments | Out-Host
    return [int]$LASTEXITCODE
}

$prepareExit = Invoke-CheckedHelper -Arguments @(
    '-PrepareEvent',
    '-ManifestPath', $manifestPath,
    '-ExpectedManifestSha256', $expectedManifestSha256,
    '-ProductionApkPath', $productionApkPath,

    '-ExpectedHelperSha256', $expectedHelperSha256,
    '-EvidenceDirectory', $eventDirectory)

$beforeExit = Invoke-CheckedHelper -Arguments @(
    '-CollectReadback',
    '-ExpectedHelperSha256', $expectedHelperSha256,
    '-EvidenceDirectory', $eventDirectory,
    '-CollectionPhase', 'Before',
    '-Serial', '15e84958',
    '-AdbPath', 'adb',
    '-ApkSignerPath', 'apksigner')
if ($beforeExit -ne 0) { throw ('RAW_LIVE_BEFORE_COLLECTION_FAILED_NO_DISPATCH_' + [string]$beforeExit) }

# Exactly one dispatch.  Keep this exit code for classification, but always
# perform the same-event after/recovery collection before deciding anything.
$dispatchExit = Invoke-CheckedHelper -Arguments @(
    '-Dispatch',
    '-ManifestPath', $manifestPath,
    '-ExpectedManifestSha256', $expectedManifestSha256,
    '-ProductionApkPath', $productionApkPath,
    '-ExpectedProductionApkSha256', $expectedProductionApkSha256,
    '-TestApkPath', $testApkPath,
    '-ExpectedTestApkSha256', $expectedTestApkSha256,
    '-ExpectedHelperSha256', $expectedHelperSha256,
    '-EvidenceDirectory', $eventDirectory)

$afterExit = Invoke-CheckedHelper -Arguments @(
    '-CollectReadback',
    '-ExpectedHelperSha256', $expectedHelperSha256,
    '-EvidenceDirectory', $eventDirectory,
    '-CollectionPhase', 'After',
    '-Serial', '15e84958',
    '-AdbPath', 'adb',
    '-ApkSignerPath', 'apksigner')

$verifyExit = Invoke-CheckedHelper -Arguments @(
    '-VerifyOutcome',
    '-EvidenceDirectory', $eventDirectory,
    '-MetadataPath', (Join-Path $eventDirectory 'HOST_RUN_METADATA.json'),
    '-PostReadbackPath', (Join-Path $eventDirectory 'post-readback.json'))

if ($dispatchExit -ne 0) { throw ('RAW_LIVE_DISPATCH_OBSERVED_NONZERO_NO_REDISPATCH_' + [string]$dispatchExit) }
if ($afterExit -ne 0) { throw ('RAW_LIVE_AFTER_COLLECTION_NOT_PROVEN_NO_REDISPATCH_' + [string]$afterExit) }
if ($verifyExit -ne 0) { throw ('RAW_LIVE_ACCEPTANCE_NOT_PROVEN_NO_REDISPATCH_' + [string]$verifyExit) }

Write-Output 'RAW_LIVE_RETRIES=0'
