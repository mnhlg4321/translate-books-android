function Assert-CommandHash {
    if ($expectedCommandSha256 -notmatch '^[0-9a-fA-F]{64}$') {
        throw 'COMMAND_SELF_HASH_EXPECTATION_MISSING_OR_INVALID_STOP'
    }
    if (-not (Test-Path -LiteralPath $commandPath -PathType Leaf)) {
        throw 'COMMAND_SELF_FILE_MISSING_STOP'
    }
    $item = Get-Item -LiteralPath $commandPath -Force
    if ($item.PSIsContainer -or (($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0)) {
        throw 'COMMAND_SELF_PATH_REPARSE_OR_NOT_REGULAR_STOP'
    }
    $resolved = [IO.Path]::GetFullPath((Resolve-Path -LiteralPath $commandPath).Path)
    if (-not [StringComparer]::OrdinalIgnoreCase.Equals($commandPath, $resolved)) {
        throw 'COMMAND_SELF_PATH_RESOLUTION_MISMATCH_STOP'
    }
    $actual = (Get-FileHash -LiteralPath $commandPath -Algorithm SHA256).Hash
    if ($actual -cne $expectedCommandSha256.ToUpperInvariant()) {
        throw 'COMMAND_SELF_HASH_MISMATCH_STOP'
    }
}

function Assert-ImmutableFile {
    param(
        [Parameter(Mandatory = $true)][string]$Path,
        [Parameter(Mandatory = $true)][string]$ExpectedSha256,
        [Parameter(Mandatory = $true)][string]$Label
    )
    if ($ExpectedSha256 -notmatch '^[0-9a-fA-F]{64}$') { throw ($Label + '_HASH_EXPECTATION_INVALID_STOP') }
    $literal = [IO.Path]::GetFullPath($Path)
    if (-not (Test-Path -LiteralPath $literal -PathType Leaf)) { throw ($Label + '_MISSING_STOP') }
    $item = Get-Item -LiteralPath $literal -Force
    if ($item.PSIsContainer -or (($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0)) { throw ($Label + '_PATH_REPARSE_OR_NOT_REGULAR_STOP') }
    $resolved = [IO.Path]::GetFullPath((Resolve-Path -LiteralPath $literal).Path)
    if (-not [StringComparer]::OrdinalIgnoreCase.Equals($literal, $resolved)) { throw ($Label + '_PATH_RESOLUTION_MISMATCH_STOP') }
    $actual = (Get-FileHash -LiteralPath $literal -Algorithm SHA256).Hash
    if ($actual -cne $ExpectedSha256.ToUpperInvariant()) { throw ($Label + '_HASH_MISMATCH_STOP') }
    return $actual
}

    [void](Assert-ImmutableFile -Path $BackupPath -ExpectedSha256 $ExpectedSha256 -Label ($Label + '_BACKUP'))
}

function Assert-HelperHash {
    param([Parameter(Mandatory = $true)][string]$Path,
          [Parameter(Mandatory = $true)][string]$ExpectedSha256)
    if ($ExpectedSha256 -notmatch '^[0-9a-fA-F]{64}$') { throw 'COMMAND_HELPER_HASH_EXPECTATION_INVALID_STOP' }
    $literalPath = [IO.Path]::GetFullPath($Path)
    if (-not (Test-Path -LiteralPath $literalPath -PathType Leaf)) { throw 'COMMAND_HELPER_MISSING_STOP' }
    $item = Get-Item -LiteralPath $literalPath -Force
    if ($item.PSIsContainer -or (($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0)) {
        throw 'COMMAND_HELPER_NOT_REGULAR_STOP'
    }
    $resolvedPath = [IO.Path]::GetFullPath((Resolve-Path -LiteralPath $literalPath).Path)
    if (-not [StringComparer]::OrdinalIgnoreCase.Equals($literalPath, $resolvedPath)) { throw 'COMMAND_HELPER_PATH_SWAP_STOP' }
    $actual = (Get-FileHash -LiteralPath $literalPath -Algorithm SHA256).Hash.ToLowerInvariant()
    if ($actual -cne $ExpectedSha256.ToLowerInvariant()) { throw 'COMMAND_HELPER_HASH_MISMATCH_STOP' }
}
